" + "Consulta realizada diretamente na TRON Mainnet."; walletData.innerHTML = `
Rede: ${network}
Ativo: ${token}
Padrão: TRC-20
Saldo: ${balance} USDT
Endereço: ${address}
Contrato: ${contract}
Estado: Mainnet
`; message.innerHTML = "Consulta concluída com sucesso."; } catch (error) { console.error( "USDTMZ WALLET ERROR:", error ); walletStatus.innerText = "Não foi possível consultar a blockchain."; message.innerHTML = "Erro:

" + "ID: " + data.withdrawal.withdrawal_id + "
