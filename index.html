async function walletInfo() { const message = document.getElementById("walletMessage"); const walletData = document.getElementById("walletData"); const walletStatus = document.getElementById("walletStatus"); const button = document.getElementById("walletButton"); message.style.display = "block"; message.innerText = "A consultar a TRON Mainnet..."; button.disabled = true; try { const response = await fetch("/api/wallet"); const walletDataResponse = await response.json(); if ( !response.ok || !walletDataResponse.success ) { throw new Error( walletDataResponse.error || "Não foi possível obter a carteira." ); } const wallet = walletDataResponse.wallet; if (!wallet) { throw new Error( "Nenhuma carteira configurada." ); } /* * O endereço e a rede vêm do servidor. */ const address = wallet.address; if (!address) { throw new Error( "A carteira não possui endereço configurado." ); } /* * Consulta diretamente a blockchain. */ const responseBlockchain = await fetch( "/api/blockchain?address=" + encodeURIComponent(address) ); const blockchain = await responseBlockchain.json(); if ( !responseBlockchain.ok || !blockchain.success ) { throw new Error( blockchain.error || "A TRON Mainnet não respondeu." ); } /* * Dados oficiais retornados pela blockchain. */ const network = blockchain.network || "TRON Mainnet"; const token = blockchain.token || "USDT"; const balance = blockchain.balance || "0.000000"; const contract = blockchain.contract || "Não informado"; /* * Atualiza a interface. */ walletStatus.innerHTML = "Blockchain conectada
" + "Consulta realizada diretamente na TRON."; walletData.innerHTML = `
Rede: ${network}
Ativo: ${token}
Padrão: TRC-20
Saldo: ${balance} USDT
Endereço: ${address}
Contrato: ${contract}
Estado: Mainnet
`; message.innerHTML = "" + "Consulta concluída com sucesso." + ""; } catch (error) { console.error( "USDTMZ WALLET ERROR:", error ); walletStatus.innerText = "Não foi possível consultar a blockchain."; message.innerHTML = "Erro:
" + error.message; } finally { button.disabled = false; } }
