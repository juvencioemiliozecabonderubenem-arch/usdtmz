async function requestWithdrawal() { const addressInput = document.getElementById("withdrawalAddress"); const amountInput = document.getElementById("withdrawalAmount"); const button = document.getElementById("withdrawalButton"); const message = document.getElementById("withdrawalMessage"); if (!addressInput || !amountInput || !button || !message) { console.error("Elementos da retirada não encontrados."); return; } const address = addressInput.value.trim(); const amount = amountInput.value.trim(); /* ========================= VALIDAR ENDEREÇO TRON ========================= */ if (!validateTronAddress(address)) { message.textContent = "Endereço TRON inválido."; return; } /* ========================= VALIDAR VALOR USDT ========================= */ if (!/^\d+(\.\d{1,6})?$/.test(amount)) { message.textContent = "Informe um valor USDT válido, com até 6 casas decimais."; return; } const numericAmount = Number(amount); if ( !Number.isFinite(numericAmount) || numericAmount <= 0 ) { message.textContent = "Informe um valor USDT válido."; return; } /* ========================= LIMITE ========================= */ if (numericAmount > 1000000) { message.textContent = "O valor máximo por retirada é 1.000.000 USDT."; return; } /* ========================= PROCESSAR NO BACKEND ========================= */ button.disabled = true; message.textContent = "A criar solicitação de retirada..."; try { const response = await fetch( "/api/withdrawal", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ destination_address: address, amount: amount }) } ); const data = await response.json(); if ( !response.ok || !data.success ) { throw new Error( data.error || "Não foi possível criar a retirada." ); } const withdrawal = data.withdrawal || data; const withdrawalId = withdrawal.withdrawal_id || data.withdrawal_id || ""; const status = String( withdrawal.status || data.status || "PENDING" ).toUpperCase(); /* ========================= TEXTO DO ESTADO ========================= */ let statusText = "Pedido recebido. Aguardando autorização."; switch (status) { case "PENDING": statusText = "Aguardando autorização."; break; case "AUTHORIZED": statusText = "Retirada autorizada."; break; case "READY_FOR_SECURE_SIGNING": statusText = "Transação pronta para assinatura segura."; break; case "PROCESSING": statusText = "Retirada em processamento."; break; case "BROADCASTED": statusText = "Transação transmitida para a TRON."; break; case "COMPLETED": statusText = "Retirada confirmada na TRON."; break; case "FAILED": statusText = "A retirada falhou."; break; default: statusText = "Estado: " + status; break; } /* ========================= TXID ========================= */ const txHash = withdrawal.tx_hash || data.tx_hash || ""; let txHtml = ""; if (txHash) { txHtml = "
TXID: " + escapeHtml(txHash); } /* ========================= RESULTADO ========================= */ message.innerHTML = "Retirada registada.
" + "ID: " + escapeHtml(withdrawalId) + "
" + "Rede: TRON Mainnet" + "
" + "Ativo: USDT TRC-20" + "
" + "Destino: " + escapeHtml(address) + "
" + "Valor: " + escapeHtml( String( withdrawal.amount ?? amount ) ) + " USDT" + "
" + "Estado: " + escapeHtml(statusText) + txHtml; /* * Só limpamos os campos depois * que o backend confirmou a criação. */ addressInput.value = ""; amountInput.value = ""; } catch (error) { console.error( "USDTMZ WITHDRAWAL ERROR:", error ); message.innerHTML = "Erro:
" + escapeHtml( error.message || "Não foi possível criar a retirada." ); } finally { button.disabled = false; } } /* ========================= VALIDAR ENDEREÇO TRON ========================= */ function validateTronAddress(address) { return /^T[1-9A-HJ-NP-Za-km-z]{33}$/.test( String(address || "").trim() ); } /* ========================= ESCAPAR HTML ========================= */ function escapeHtml(value) { return String(value) .replaceAll("&", "&") .replaceAll("<", "<") .replaceAll(">", ">") .replaceAll('"', """) .replaceAll("'", "'"); } /* ========================= BOTÃO DE RETIRADA ========================= */ const withdrawalButton = document.getElementById( "withdrawalButton" ); if (withdrawalButton) { withdrawalButton.addEventListener( "click", requestWithdrawal ); }
